% Tiny standalone source that produces a placeholder logo.pdf.
% Build with:  pdflatex -output-directory=. logo-source.tex
% Then rename the output to logo.pdf (or replace logo.pdf with your real logo).
\documentclass[border=2pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{shapes.geometric}
\definecolor{logoA}{HTML}{1F4E79}
\definecolor{logoB}{HTML}{E08E0B}
\begin{document}
\begin{tikzpicture}
  \node[regular polygon, regular polygon sides=6, draw=logoA, line width=1.5pt, minimum size=18mm, fill=logoA!10] (h) {};
  \node[font=\bfseries\Large, color=logoA] at (h.center) {IS};
  \node[anchor=west, font=\bfseries\Large, color=logoA] at (h.east) {Industrial};
  \node[anchor=west, font=\bfseries\Large, color=logoB] at ([yshift=-5mm]h.east) {Security};
\end{tikzpicture}
\end{document}
