\sectionoverviewframe

\learningobjectives{
  \item Extract a ladder-logic program from a PLC and from a project file.
  \item Recognise vendor-specific block formats (Siemens, Rockwell, CODESYS).
  \item Decompile or visualise ladder logic and identify what the program does.
  \item Spot logic-bomb and stealth modifications (\`a la Stuxnet).
  \item Apply integrity controls to prevent unnoticed code change.
}

\begin{importantbox}{The blind spot}
The PLC runs the \emph{compiled} blocks. The vendor tool shows the \emph{project}. If an attacker changes only the runtime, the project still looks fine and only RE catches it.
\end{importantbox}
\source{Stuxnet (2010) is the canonical example -- HMI fed clean data while the PLC ran tampered logic.}

\begin{frame}{Rockwell .L5K and .ACD}
\begin{itemize}
  \item \texttt{.acd} is a binary container; \texttt{.l5k} / \texttt{.l5x} is the printable export.
  \item \texttt{l5k\_parser} (community) reconstructs Structured Text from the export.
  \item Watch for \textbf{add-on instructions (AOIs)} -- compiled inside the project; their internals are hidden in the GUI.
  \item Tag aliases obscure semantics: \texttt{Bool15} may secretly be ``open\_valve''.
\end{itemize}
\begin{tipbox}{Decompile then re-render}
Round-trip the export through RSLogix; what visually matches the original is faithful, what diverges is your target.
\end{tipbox}
\end{frame}

\section{What to Look For}

\begin{frame}{The Stuxnet Pattern}
\centering
\begin{tikzpicture}[node distance=4mm]
  \node[isbox, fill=isAttack!10, draw=isAttack, minimum width=110mm, font=\small] (a) {1.\ Pristine OB1 / main loop sets up normal control};
  \node[isbox, fill=isAttack!10, draw=isAttack, minimum width=110mm, font=\small, below=of a] (b) {2.\ Hidden OB35 (cyclic) checks for the target signature (frequency, set-point)};
  \node[isbox, fill=isAttack!10, draw=isAttack, minimum width=110mm, font=\small, below=of b] (c) {3.\ On match: record current set-points, replace with malicious sequence};
  \node[isbox, fill=isAttack!10, draw=isAttack, minimum width=110mm, font=\small, below=of c] (d) {4.\ When operator reads back, recorded set-points are returned (HMI lied)};
\end{tikzpicture}
\source{Langner, R., \emph{To Kill a Centrifuge}, The Langner Group, 2013.}
\end{frame}

\begin{frame}{Logic Bomb Indicators}
\begin{itemize}
  \item Unused inputs that are read but never appear in the rung logic.
  \item Time- or counter-based branches that never fire under normal data.
  \item Blocks named after vendor utilities (``OEM\_diag'', ``factory\_test'') outside the official block list.
  \item Constants whose values match \emph{exactly} known process set-points.
  \item Calls into reserved or vendor-locked memory areas.
  \item Discrepancy between project-file timestamps and on-PLC block CRCs.
\end{itemize}
\begin{warnbox}
A missing CRC, or a CRC that does not match the project file, is more often vendor sloppiness than malice -- but it deserves a written investigation.
\end{warnbox}
\end{frame}

Cross-reference \emph{every} symbol against where it is read and where it is written. Symbols that are only \emph{written} (never read) are smoke.
\end{notebox}

\section{Defending the Code}

\begin{frame}{Defensive Controls}
\centering
\begin{tikzpicture}[node distance=4mm]
  \node[isbox, fill=isOT!15, draw=isOT, minimum width=110mm, font=\small] (a) {Signed firmware + signed program downloads (vendor support varies)};
  \node[isbox, fill=isOT!15, draw=isOT, minimum width=110mm, font=\small, below=of a] (b) {PLC key-switch in ``RUN'' position; ``PROG'' requires physical access};
  \node[isbox, fill=isOT!15, draw=isOT, minimum width=110mm, font=\small, below=of b] (c) {Continuous block-checksum monitoring (Claroty, Dragos, Nozomi modules)};
  \node[isbox, fill=isOT!15, draw=isOT, minimum width=110mm, font=\small, below=of c] (d) {Version-controlled project files with CI \& review};
  \node[isbox, fill=isOT!15, draw=isOT, minimum width=110mm, font=\small, below=of d] (e) {Engineering workstation hardened; no Internet, MFA, application allow-listing};
\end{tikzpicture}
\end{frame}

\begin{frame}{Disclosure Considerations}
\begin{itemize}
  \item Findings on production logic touch \textbf{safety} -- coordinate with process safety, not just IT.
  \item Vendor PSIRT first; CERT/CC second; public publication only after fix.
  \item Sharing tags or constants from real plants risks identifying the asset owner -- redact heavily.
  \item Some jurisdictions (US DMCA \S1201, EU CRA research safe-harbour, German \S202c StGB) constrain RE; consult counsel.
\end{itemize}
\begin{warnbox}
Logic from a production PLC you do not own is contaminated evidence. No serious vendor will accept it.
\end{warnbox}
\end{frame}

%% Frame 1: Standards anchor -- IEC 61131-3 languages, 62443-4-2 CR 3.4, compile chain
\begin{frame}{Standards anchor for control-logic RE}
\centering
\begin{tikzpicture}[node distance=2mm, scale=0.85, transform shape]
  \node[isbox, fill=isOT!15, draw=isOT, minimum width=110mm, font=\small, align=left, text width=104mm] (a)
    {\textbf{IEC 61131-3} \ Five PLC languages you reverse: \textbf{LD} ladder, \textbf{IL} instruction list, \textbf{ST} structured text, \textbf{FBD}, \textbf{SFC}};
  \node[isbox, fill=isPLC!15, draw=isPLC, minimum width=110mm, font=\small, below=of a, align=left, text width=104mm] (b)
    {\textbf{POU model} \ Programs / function blocks / functions over \texttt{\%I}, \texttt{\%Q}, \texttt{\%M} located variables --- the RE target language};
  \node[isbox, fill=isAccent!15, draw=isAccent, minimum width=110mm, font=\small, below=of b, align=left, text width=104mm] (c)
    {\textbf{62443-4-2 CR 3.4} \ Software / firmware \emph{integrity} --- the control whose absence makes silent logic swaps possible};
  \node[isbox, fill=isAttack!15, draw=isAttack, minimum width=110mm, font=\small, below=of c, align=left, text width=104mm] (d)
    {\textbf{Compile chain} \ Source $\to$ vendor / IEC compiler $\to$ on-PLC artefact (bytecode or native) --- divergence is the attack surface};
\end{tikzpicture}
\par\vspace{2mm}
{\footnotesize 61131-3 fixes \emph{what} the logic says; CR 3.4 is \emph{whether} what runs is what was signed. RE bridges the gap when CR 3.4 is missing.}
\source{IEC 61131-3:2013 \textbar{} IEC 62443-4-2:2019, CR 3.4 (software/firmware integrity) \textbar{} PLCopen, \emph{IEC 61131-3 a Standardized Programming Language}, 2014.}
\note{
  This is the spine of the section. Three load-bearing points. First, 61131-3 is the language family --- a Master audience already knows LD/ST; the move here is that all five map onto the same POU/located-variable model, so reversing one is reversing all. Second, CR 3.4 is the single component requirement that, when implemented (signed downloads, verified boot), would have stopped Stuxnet-class logic swaps; its absence is precisely what creates work for an RE analyst. Third, the compile chain is the reason runtime and project diverge --- hammer that the PLC executes the compiled artefact, never the source. Pre-empt: students conflate ``the standard'' with ``the vendor tool'' --- 61131-3 is portable, the artefact format is not. Transition: ``So what exactly does that compile chain produce, and at which layer do you reverse it?''
}
\end{frame}

\par\vspace{1mm}
{\footnotesize \textbf{Reverse at the layer you can reach:} source if you have the project; \texttt{POUS.c} if you catch the build; \textbf{ELF / Ghidra} if all you have is the device.}
\par\vspace{1mm}
\begin{notebox}{Why the C layer matters}
OpenPLC and many soft-PLCs compile IEC code to C, then to native code --- the running logic is a stripped shared object, not bytecode. \texttt{LOCATED\_VARIABLES.h} is the Rosetta stone mapping \texttt{\%QX0.0} to a C symbol and a memory offset.
\end{notebox}
\source{T.\ R.\ Alves, OpenPLC / MatIEC build pipeline, 2018 \textbar{} Beremiz/MatIEC project documentation (IEC 61131-3 to C), 2013--.}
\note{
  Master-level depth: this frame says where your tools attach. Walk left to right, then make the key point --- as you move down the chain you lose symbols and gain ground truth. At the source layer you have names; at POUS.c you still have readable C and the located-variables header; at the ELF you have offsets and Ghidra. The LOCATED\_VARIABLES.h call-out is the practical gem: it is the deterministic map from \%IX/\%QX to C variables, so if you can grab it from a build artefact you skip most of the static-analysis pain. Tie back to Frame 1: this is the compile chain that CR 3.4 is supposed to protect end to end. Note the contrast with Siemens MC7 (bytecode) covered later --- soft-PLCs go all the way to native, hard PLCs often stop at bytecode; the RE workflow differs. Transition: ``Once you can read the logic at some layer, the real question is not syntax but intent.''
}
\end{frame}

%% Frame 3: Reconstructing control intent (worked seal-in latch example)
\begin{frame}[fragile]{Reconstructing control intent: a seal-in latch}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{lstlisting}[basicstyle=\ttfamily\tiny\color{slideTitle}]
Start AT %IX0.0 : BOOL;
Stop  AT %IX0.1 : BOOL;(*NC*)
Guard1 AT %IX0.2 : BOOL;
Guard2 AT %IX0.3 : BOOL;
Motor AT %QX0.0 : BOOL;
Lamp  AT %QX0.1 : BOOL;
Motor := (Start OR Motor)
   AND Stop AND Guard1
   AND Guard2;
Lamp  := Motor;
\end{lstlisting}
\end{column}
\begin{column}{0.5\textwidth}
\scriptsize
\textbf{Reconstructed I/O map}
\begin{itemize}\setlength{\itemsep}{1pt}
  \item \texttt{\%IX0.0} start PB \,(momentary)
  \item \texttt{\%IX0.1} stop PB \,(NC, fail-safe)
  \item \texttt{\%IX0.2/.3} two guard switches
  \item \texttt{\%QX0.0} motor; \texttt{\%QX0.1} run lamp
\end{itemize}
\textbf{Interlock logic}
\begin{itemize}\setlength{\itemsep}{1pt}
  \item \texttt{(Start OR Motor)} = seal-in latch
  \item Stop+Guard1+Guard2 all TRUE to run (de-energise-to-stop)
\end{itemize}
\end{column}
\end{columns}
\begin{notebox}{Intent, not syntax, is the goal}
The win is naming the \emph{behaviour}: ``self-holding motor start, fail-safe stop, two guard interlocks'' --- a decompile you cannot state in one safety sentence is transcribed, not reversed.
\end{notebox}
\source{IEC 61131-3:2013 (seal-in / latch pattern) \textbar{} repo lab \texttt{labs/\_stack/scripts/ladder/conveyor.st}.}
\note{
  The pedagogical pivot of the section: decompilation is necessary but not sufficient. Work the example live --- read the rung, then force the class to articulate intent. Key teaching beats: (1) the \texttt{(Start OR Motor)} term is the classic seal-in/latch --- once Motor is TRUE it holds itself until a series condition drops; recognising this idiom instantly is the RE skill. (2) Stop is normally-closed and ANDed, so a broken wire stops the motor --- de-energise-to-stop is a safety design choice you must call out, because an attacker who flips it to normally-open removes the fail-safe with a one-character change. (3) Two guards in series = both must be closed; this is a safety interlock, not arbitrary logic. The notebox is the assessment criterion: if a student hands back ST that is correct but cannot say what it is FOR, mark it incomplete. Transition: ``Now imagine logic that LOOKS like this rung but hides a payload.''
}
\end{frame}

%% Frame 4: Logic-bomb / malicious-logic patterns + baseline-diff detection
\begin{frame}{Logic bombs: hiding payloads in PLC code}
\centering
\begin{tikzpicture}[node distance=2mm, scale=0.85, transform shape]
  \node[isbox, fill=isAttack!10, draw=isAttack, minimum width=110mm, font=\small, align=left, text width=104mm] (a)
    {\textbf{Counter-gated} \ Fires only after N scans --- invisible in a short bench test};
  \node[isbox, fill=isAttack!10, draw=isAttack, minimum width=110mm, font=\small, below=of a, align=left, text width=104mm] (b)
    {\textbf{Time-gated} \ Keyed on RTC/date --- Stuxnet armed only on a process match};
  \node[isbox, fill=isAttack!10, draw=isAttack, minimum width=110mm, font=\small, below=of b, align=left, text width=104mm] (c)
    {\textbf{Hidden OB / FC} \ Cyclic block (OB35) or rare function outside the block list};
  \node[isbox, fill=isAttack!10, draw=isAttack, minimum width=110mm, font=\small, below=of c, align=left, text width=104mm] (d)
    {\textbf{Set-point mimicry} \ Replays true values on read --- HMI normal while output diverges};
\end{tikzpicture}
\par\vspace{1mm}
\begin{importantbox}{Detection: baseline-diff against a signed golden project}
None of the above survives a byte-level diff of every block CRC against a \emph{signed} golden project archived off-line. Compare on-PLC blocks to the signed baseline, not the engineering-workstation copy --- the WS can be tampered too.
\end{importantbox}
\source{R.\ Langner, \emph{To Kill a Centrifuge}, 2013 \textbar{} N.\ Falliere et al., \emph{W32.Stuxnet Dossier} v1.4, Symantec, 2011 \textbar{} IEC 62443-4-2:2019 CR 3.4.}
\note{
  Synthesis of attack + defence. The four rows are a taxonomy of how malicious logic stays dormant: gating (counter or time) defeats short bench tests, hidden blocks defeat a casual block-list review, and set-point mimicry defeats the operator's eyes --- this last one is the Stuxnet move and ties straight back to Section's earlier ``HMI lied'' frame, so cross-reference it. The importantbox is the only durable answer: you cannot eyeball your way out of a logic bomb, you need a cryptographically signed golden baseline kept off the engineering workstation, and an automated CRC/byte diff of every block on the live PLC. Stress the trust-anchor subtlety: diffing against the WS copy is worthless if the WS was the entry vector --- the baseline must be signed and stored off-line. Tie the whole thing to CR 3.4 from Frame 1: signed downloads plus verified boot would make most of these undetectable-by-eye patterns detectable-by-policy. Close: this is why integrity monitoring, not just access control, defines a mature OT program.
}
\end{frame}


\begin{frame}{Summary -- Section \sectionNumber}
\begin{takeawaybox}{Ladder RE for ICS}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item PLCs run compiled blocks, not the project source -- they can diverge silently.
  \item Project-file and runtime acquisition each catch different attacks.
  \item Vendor-specific tools (snap7, l5k-parser, CODESYS) are the workhorses.
  \item Stuxnet-style logic bombs hide in unused inputs, time bombs, and CRC mismatches.
  \item Defence is signed downloads, key-switch, continuous checksum monitoring.
\end{itemize}
\end{takeawaybox}
\end{frame}
