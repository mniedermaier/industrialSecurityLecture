\sectionoverviewframe

\learningobjectives{
  \item Distinguish authorised testing, red-team exercises, and unauthorised access -- legally and ethically.
  \item Build a contained lab in which destructive testing is safe.
  \item Apply standard ICS-attack techniques (reconnaissance, protocol abuse, logic injection) with full attribution.
  \item Produce findings that the asset owner can act on without re-doing the work.
}

\section{Ethics and Scope}

\begin{frame}{Authorised vs.\ Unauthorised}
\centering
\begin{tikzpicture}[node distance=4mm]

  \node[isbox, fill=isAccent!15, draw=isAccent, minimum width=38mm, font=\scriptsize, align=center, right=of auth] (red) {Authorised red team\\(internal-blind)};
  \node[isbox, fill=isAttack!15, draw=isAttack, minimum width=38mm, font=\scriptsize, align=center, right=of red] (unauth) {Unauthorised access\\(criminal)};
\end{tikzpicture}
\vspace{2mm}
\begin{importantbox}{Paper, not handshake}
``Verbal'' permission is no permission. A signed Rules of Engagement document with named approvers, scope, timing, kill-switch, and reporting path is the gating artefact for any engagement.
\end{importantbox}
\source{NIST SP 800-115 \emph{Technical Guide to Information Security Testing}; PTES Technical Guidelines.}
\end{frame}

\begin{frame}{Legal Surface (EU/DE/US -- partial)}
\begin{itemize}
  \item \textbf{Germany:} \S 202a / 202c StGB criminalise unauthorised access and \emph{preparation tools}; commercial pentesting needs written authorisation.
  \item \textbf{EU CRA (2024/2847):} research safe-harbour for vulnerability disclosure under coordinated terms.
  \item \textbf{US CFAA \& DMCA \S1201:} authorisation contracts grant access; safe-harbour for security research since 2015 rulemaking.
  \item \textbf{ISO/IEC 29147 \& 30111:} normative guidance for coordinated disclosure.
\end{itemize}
\begin{warnbox}
Cross-border engagements need both legal opinions. A US-headquartered firm testing a German plant follows the strictest applicable law.
\end{warnbox}
\source{StGB \S 202a, \S 202c; Directive (EU) 2022/2555; Regulation (EU) 2024/2847; ISO/IEC 29147:2018, 30111:2019.}
\end{frame}

\section{Tactics}

\begin{frame}{Reconnaissance in OT (passive first)}
\begin{itemize}
  \item ARP / LLDP / CDP listening to enumerate devices.
  \item Passive Zeek capture for 24h to build a baseline.
  \item Vendor banners on web UIs (read-only HTTP GET / robots.txt).
  \item Project files on file shares (engineering WS, contractor laptops).
\end{itemize}
\begin{warnbox}{Active scans on the bench}
Nmap with \texttt{--max-rate} and only NSE scripts known safe for the target. Run once, capture, then decide if a deeper probe is worth the risk.
\end{warnbox}
\source{ICS-CERT (CISA) recommended practices; Dragos \emph{ICS Knowledge Pack}.}
\end{frame}

\section{Reporting}

\begin{frame}{Report Structure}
\centering
\begin{tikzpicture}[node distance=1mm]
  \node[isbox, fill=isOT!15, draw=isOT, minimum width=110mm, font=\footnotesize, minimum height=5mm] (a) {1.\ Executive summary -- 1 page, plain language};
  \node[isbox, fill=isOT!15, draw=isOT, minimum width=110mm, font=\footnotesize, minimum height=5mm, below=of a] (b) {2.\ Scope, ROE, attestations};
  \node[isbox, fill=isOT!15, draw=isOT, minimum width=110mm, font=\footnotesize, minimum height=5mm, below=of b] (c) {3.\ Findings, each: title, CVSS-E, evidence, repro};
  \node[isbox, fill=isOT!15, draw=isOT, minimum width=110mm, font=\footnotesize, minimum height=5mm, below=of c] (d) {4.\ Recommendations, sized this-quarter / this-year};
  \node[isbox, fill=isOT!15, draw=isOT, minimum width=110mm, font=\footnotesize, minimum height=5mm, below=of d] (e) {5.\ Appendix: PCAPs, configs, raw logs};
\end{tikzpicture}
\vspace{1mm}
\begin{tipbox}{Reproducibility}
A finding without a step-by-step repro is theatre. If the asset owner cannot reproduce it after a week, they will not trust the rest of the report.
\end{tipbox}
\source{NIST SP 800-115, \S 8 \emph{Reporting}, 2008 \textbar{} PTES \emph{Reporting} section; OWASP Web Security Testing Guide v4, \emph{Reporting} chapter.}
\end{frame}

\begin{frame}{Common Reporting Pitfalls}
\begin{itemize}
  \item Listing CVEs without saying which versions are actually in the plant.
  \item Confusing ``vulnerable'' with ``exploitable in this environment''.
  \item Recommendations without owners or timelines.
  \item Findings ranked by CVSS only, not by process-impact.
  \item Reusing screenshots between clients (legal disaster).
\end{itemize}
\begin{warnbox}
The asset owner's safety engineers should be able to read the report and say whether the test \emph{could have} caused a process upset. If not, your scope was wrong.
\end{warnbox}
\source{NIST SP 800-115, \S 8 (reporting pitfalls), 2008 \textbar{} PTES Technical Guidelines, \emph{Reporting}.}
\end{frame}

\begin{frame}{Framework anchor -- ATT\&CK for ICS \& the ICS Kill Chain}
\centering
\begin{tikzpicture}[node distance=2mm, scale=0.85, transform shape]
  \node[isbox, fill=isIT!15, draw=isIT, minimum width=120mm, font=\footnotesize, align=left, text width=114mm] (a)
    {\textbf{Recon \& Initial Access} \ ATT\&CK \emph{Initial Access} (T0817 Drive-by, T0866 Remote Services) $\;\equiv\;$ Kill Chain \textbf{Stage 1}};
  \node[isbox, fill=isAccent!15, draw=isAccent, minimum width=120mm, font=\footnotesize, align=left, below=of a, text width=114mm] (b)
    {\textbf{Pivot \& Cross iDMZ} \ ATT\&CK \emph{Lateral Movement} (T0859 Valid Accounts, T0818 Eng.\ Workstation Compromise) $\;\equiv\;$ end of \textbf{Stage 1}};
  \node[isbox, fill=isOT!15, draw=isOT, minimum width=120mm, font=\footnotesize, align=left, below=of b, text width=114mm] (c)
    {\textbf{Logic \& Command} \ ATT\&CK \emph{Impair Process Control} (T0836 Mod.\ Param., T0843 Program Download) $\;\equiv\;$ \textbf{Stage 2}};
  \node[isbox, fill=isAttack!15, draw=isAttack, minimum width=120mm, font=\footnotesize, align=left, below=of c, text width=114mm] (d)
    {\textbf{Disable Safety \& Effect} \ ATT\&CK \emph{Inhibit Response} (T0881 Service Stop) \& \emph{Impact} (T0827 Loss of Control, T0880 Loss of Safety) $\;\equiv\;$ \textbf{Stage 2}};
\end{tikzpicture}
\par\vspace{2mm}
\begin{notebox}{Two lenses, one engagement}
ATT\&CK for ICS names \emph{techniques}; the Assante/Lee kill chain names \emph{stages}. Map every authorised action to both: the technique tells the blue team what to detect, the stage tells them where they still have leverage.
\end{notebox}
\source{MITRE ATT\&CK for ICS, \texttt{attack.mitre.org/matrices/ics/} \textbar{} M.\ Assante \& R.\ Lee, \emph{The ICS Cyber Kill Chain}, SANS, 2015.}
\note{
  This is the anchor frame for the offensive section: two complementary frameworks the rest of the deck maps onto. Read the four rows top-down as the attacker's arc and stress that each row carries both an ATT\&CK technique ID and a kill-chain stage. ATT\&CK for ICS is the technique taxonomy (the matrix runs Initial Access $\to$ Execution $\to$ Persistence $\to$ Privilege Escalation $\to$ Evasion $\to$ Discovery $\to$ Lateral Movement $\to$ Collection $\to$ Command \& Control, then the three ICS-distinctive tactics: \emph{Inhibit Response Function}, \emph{Impair Process Control}, and \emph{Impact}). The Assante/Lee chain is the staged model: Stage 1 is the IT intrusion (recon, weaponise, deliver, exploit, install, C2, act) and Stage 2 is the ICS attack-development-and-execution stage (develop, test, deliver, install/modify, execute the cyber-physical effect). The pedagogical point for the room: an authorised tester reports findings in ATT\&CK technique IDs because that is the language the asset owner's detection engineering team already speaks, while the kill-chain stage tells you which control layer should have caught it. Tie every later frame in this section back here.
}
\end{frame}

\centering
\begin{tikzpicture}[node distance=2mm, every node/.style={font=\scriptsize}, scale=0.82, transform shape]
  \node[isbox, fill=isIT!15, draw=isIT, minimum width=18mm, minimum height=13mm, align=center] (a) {1. Phish /\\stolen creds};
  \node[isbox, fill=isIT!15, draw=isIT, right=of a, minimum width=18mm, minimum height=13mm, align=center] (b) {2. Pivot\\IT LAN};

  \node[isbox, fill=isOT!15, draw=isOT, right=of c, minimum width=18mm, minimum height=13mm, align=center] (d) {4. Eng.\\workstation};
  \node[isbox, fill=isPLC!15, draw=isPLC, right=of d, minimum width=18mm, minimum height=13mm, align=center] (e) {5. PLC /\\logic push};
  \node[isbox, fill=isAttack!15, draw=isAttack, right=of e, minimum width=18mm, minimum height=13mm, align=center] (f) {6. Process /\\physical};
  \foreach \x/\y in {a/b, b/c, c/d, d/e, e/f}{\draw[attack] (\x) -- (\y);}
  % stage brackets
  \node[font=\tiny\bfseries, text=isIT, below=5mm of a.south, align=left] {Stage 1\\(IT intrusion)};
  \node[font=\tiny\bfseries, text=isOT, below=5mm of e.south, align=left] {Stage 2\\(eng.\,$\to$\,PLC\,$\to$\,process)};
  % detection break markers above the chain
  \node[font=\tiny, text=isNet, above=4mm of c.north, align=center, fill=isNet!12, draw=isNet, inner sep=1pt] (dx1) {EDR / NetFlow\\breaks here};
  \node[font=\tiny, text=isNet, above=4mm of e.north, align=center, fill=isNet!12, draw=isNet, inner sep=1pt] (dx2) {OT-IDS\\breaks here};
  \draw[->, isNet, thick] (dx1.south) -- (c.north);
  \draw[->, isNet, thick] (dx2.south) -- (e.north);
\end{tikzpicture}
\par\vspace{3mm}
\begin{importantbox}{Authorised testing earns its keep at the breaks}
Detection \emph{anywhere} severs the chain. An authorised engagement is most valuable when it proves the asset owner can break it at the iDMZ (Stage 1$\to$2) and again on the OT wire (Stage 2) -- not when it merely reaches the process.
\end{importantbox}
\source{Adapted from M.\ Assante \& R.\ Lee, \emph{The ICS Cyber Kill Chain}, SANS, 2015 \textbar{} MITRE ATT\&CK for ICS.}
\note{
  Walk the chain left-to-right. Stages 1--2 are bog-standard IT attack work (phishing, credential theft, lateral movement) and IT defences still apply; this is where EDR and NetFlow anomaly detection have a chance. Stage 3 is the iDMZ crossing -- the single highest-leverage detection point because it is a chokepoint the asset owner controls and monitors. Once the attacker lands on the engineering workstation (Stage 4) and pushes logic to the PLC (Stage 5), only OT-aware controls -- a protocol-aware OT-IDS watching for S7 program-download or Modbus writes -- can see it. The two green markers are the message of the slide: a good authorised test does not just demonstrate it can reach Stage 6, it demonstrates exactly where the blue team's existing controls do and do not sever the chain. Reinforce the strict authorised-use frame: you only run this against a contained lab or a scoped, signed engagement. Tie Stage 5 detection forward to Frame 4 (offense tied to detection). The leverage point for the room: the defender does not have to win at every stage -- breaking it once defeats the attack.
}
\end{frame}

\centering
\begin{tikzpicture}[node distance=7mm, scale=0.85, transform shape]

  \draw[attack] (core) -- (sch);

  \draw[attack] (core) -- (opc);
  \draw[attack] (opc) -- node[above, font=\tiny, yshift=0.5mm, fill=white, inner sep=1pt]{vendor-agnostic} (any);
\end{tikzpicture}
\par\vspace{2mm}
\begin{importantbox}{The framework, not the bug -- caught pre-deployment}
PIPEDREAM is modular: per-vendor modules plus an OPC UA pivot that is vendor-agnostic \emph{by construction}. Patching one CVE does not blunt it -- the capability outlives any single bug. It is the first ICS attack framework discovered \emph{before} any victim impact.
\end{importantbox}
\source{CISA \emph{AA22-103A}, ``APT Cyber Tools Targeting ICS/SCADA Devices'', 13 Apr 2022 \textbar{} Dragos, \emph{PIPEDREAM/CHERNOVITE}, Apr 2022 \textbar{} Mandiant, \emph{INCONTROLLER}, Apr 2022.}
\note{
  PIPEDREAM (Dragos's toolkit name; CHERNOVITE the actor) and INCONTROLLER (Mandiant's name) are the same modular framework, wrapped by joint advisory CISA AA22-103A. Architecture is the lesson: a core/launcher orchestrates vendor-specific modules --- \textsc{evilscholar} for Schneider M580/M340, \textsc{badomen} for Omron NX/NJ --- plus \textsc{mousehole}, an OPC UA module that pivots between devices regardless of vendor because OPC UA is the lingua franca of modern ICS. Drive the ``framework, not the bug'' point: CVE-chasing students want a patch, but a framework that abuses legitimate engineering functions does not depend on any one CVE --- close a module's bug, the architecture survives. Same modular lineage as Industroyer and TRITON, only broader and caught earlier --- the first time we got ahead of an ICS capability before deployment. For the authorised tester: emulating \emph{framework behaviour} (multi-vendor dispatch, OPC UA pivoting) tests detection better than reproducing one exploit.
}
\end{frame}

%% Frame 4: Offense tied to detection -- techniques vs what an OT-IDS sees
\begin{frame}{Offense Tied to Detection -- what the OT-IDS sees}
\centering
\renewcommand{\arraystretch}{1.15}
\scriptsize
\setlength{\tabcolsep}{3pt}
\begin{tabular}{|p{30mm}|p{16mm}|p{60mm}|}
\hline
\rowcolor{slidePanel}
\textbf{Offensive technique} & \textbf{ATT\&CK} & \textbf{Zeek / ICSNPP visibility} \\
\hline
Network scan / device enum & T0846 & \textbf{Sees} -- new conn fan-out, function-code sweep across hosts \\
\hline
Modbus coil/register write & T0855 & \textbf{Sees} -- \texttt{modbus.log} write FC 5/6/15/16 off baseline \\
\hline
S7 program download (logic) & T0843 & \textbf{Sees} -- \texttt{s7comm.log} job: download block to CPU \\
\hline
OPC UA over TLS (encrypted) & T0869 & \textbf{Blind} -- payload encrypted; only endpoint metadata \\
\hline
Spoof reporting / HMI replay & T0856 & \textbf{Partial} -- needs process-baseline model, not parsing alone \\
\hline
\end{tabular}
\par\vspace{2mm}
\begin{importantbox}{Build detection \emph{from} offense}
Every authorised technique you run is a detection-engineering test case. Map it to its ATT\&CK ID, capture the PCAP, and hand the blue team the exact Zeek signature it should -- or could not -- have produced.
\end{importantbox}
\source{MITRE ATT\&CK for ICS (T0846/T0855/T0843/T0869/T0856) \textbar{} Zeek + ICSNPP protocol analysers, \texttt{github.com/cisagov/icsnpp}.}
\note{
  Closes the offensive section by tying every technique back to the defender's instrument. Walk each row and stress the \textbf{Sees / Blind / Partial} verdict. Scanning (T0846) is loud --- Zeek logs the connection fan-out and ICSNPP flags the function-code sweep. Modbus writes (T0855) and S7 program downloads (T0843) are fully visible: ICSNPP analysers parse \texttt{modbus.log} and \texttt{s7comm.log} down to the function code and download-block job; the tester should produce those log lines and confirm an alert fires. Two hard cases are the lesson: OPC UA over TLS (T0869) is a \emph{blind spot} --- only endpoint metadata visible, which is why the PIPEDREAM \textsc{mousehole} pivot is dangerous; mitigations are endpoint logging or a sanctioned TLS-inspection point. HMI replay (T0856 --- the Stuxnet trick) is only \emph{partially} detectable from the wire: protocol parsing alone cannot tell a replayed-but-valid value from a real one, so you need a process-behaviour baseline. Land the importantbox: every authorised technique becomes a labelled detection test case --- run it, capture the PCAP, hand over the Zeek/ICSNPP signature that should have caught it (or document the gap).
}
\end{frame}


\begin{frame}{Summary -- Section \sectionNumber}
\begin{takeawaybox}{Offensive ICS, responsibly}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item Signed Rules of Engagement gate every action -- no signature, no test.
  \item Build a contained lab; never fuzz or inject on production.
  \item Standard ICS-attack patterns (recon, protocol abuse, logic injection) cover most engagements.
  \item Report in plain language with reproducible evidence and prioritised recommendations.
  \item Cross-border work follows the strictest applicable law.
\end{itemize}
\end{takeawaybox}
\end{frame}
